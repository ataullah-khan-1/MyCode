if (autoSavedPath) {
            const dir = path.dirname(autoSavedPath);
            const base = path.basename(autoSavedPath);
            const alreadyPrefixed = base.toLowerCase().startsWith(`${prefix}-`);
            if (alreadyPrefixed) {
                CSReporter.info(`${prefix} download resolved via framework auto-save: ${autoSavedPath}`);
                return autoSavedPath;
            }
            const renamed = path.resolve(dir, `${prefix}-${base}`);
            try {
                fs.renameSync(autoSavedPath, renamed);
                mgr.addDownloadedFile(renamed, path.basename(renamed), suggested);
                CSReporter.info(`${prefix} download renamed to avoid clobber: ${renamed}`);
                return renamed;
            } catch (e) {
                CSReporter.warn(`${prefix} rename failed (${(e as Error).message}); returning auto-saved path as-is`);
                return autoSavedPath;
            }
        }
