
import { readFileSync, writeFileSync } from 'node:fs';
const [input, name, output = input.replace(/\.rdl$/i, '') + '_shown.rdl'] = process.argv.slice(2);
if (!input || !name) { console.error('Usage: node unhide.mjs <input.rdl> <ItemName> [output.rdl]'); process.exit(1); }
const buf = readFileSync(input);
const utf16 = buf[0] === 0xff && buf[1] === 0xfe;
let text = (utf16 ? buf.toString('utf16le') : buf.toString('utf8')).replace(/^﻿/, '');
const start = text.search(new RegExp(`<\\w+ Name="${name}"\\s*>`));
if (start < 0) { console.error(`Item ${name} not found`); process.exit(1); }
// Walk the tags to find the item's direct <Visibility> child and its end.
const tag = /<!\[CDATA\[[\s\S]*?\]\]>|<!--[\s\S]*?-->|<(\/?)([\w:]+)[^>]*?(\/?)>/g;
tag.lastIndex = start;
let depth = 0, visStart = -1, visEnd = -1, m;
while ((m = tag.exec(text))) {
  if (!m[2]) continue;
  if (m[1]) { depth--; if (depth === 1 && m[2] === 'Visibility' && visStart >= 0 && visEnd < 0) visEnd = tag.lastIndex; if (depth === 0) break; }
  else if (!m[3]) { if (depth === 1 && m[2] === 'Visibility') visStart = m.index; depth++; }
}
if (visStart < 0) { console.error(`${name} has no Visibility setting; it is already always shown`); process.exit(1); }
console.log('Old:', text.slice(visStart, visEnd).replace(/\s+/g, ' '));
text = text.slice(0, visStart) + '<Visibility><Hidden>false</Hidden></Visibility>' + text.slice(visEnd);
writeFileSync(output, utf16 ? Buffer.concat([Buffer.from([0xff, 0xfe]), Buffer.from(text, 'utf16le')]) : text);
console.log('Written:', output);
