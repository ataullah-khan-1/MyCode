import { readFileSync } from 'node:fs';

const xml = readFileSync(process.argv[2], 'utf8');
const items = new Set(['Textbox', 'Rectangle', 'Tablix', 'Chart', 'Line', 'Image', 'Subreport']);
const tags = [...xml.matchAll(/<(\/?)(\w+)((?:\s+[\w:]+="[^"]*")*)\s*(\/?)>/g)];
const stack = [];
const nodes = [];
let bodyAt = -1;
for (const m of tags) {
  const [, close, name, attrs, self] = m;
  if (close) { stack.pop(); continue; }
  const node = { name, attr: /Name="([^"]*)"/.exec(attrs ?? '')?.[1], depth: stack.length, at: m.index, parent: stack[stack.length - 1] };
  nodes.push(node);
  if (name === 'Body' && bodyAt < 0) bodyAt = nodes.length - 1;
  if (!self) stack.push(node);
}
const valueAfter = (node, tag) => {
  const end = xml.indexOf(`</${node.name}>`, node.at);
  const part = xml.slice(node.at, end > 0 ? end : undefined);
  const depthTag = new RegExp(`<${tag}>([^<]*)</${tag}>`, 'g');
  let m;
  let best;
  while ((m = depthTag.exec(part))) { best = m[1]; break; }
  return best;
};
const body = nodes[bodyAt];
const inBody = nodes.filter((n) => n.at > body.at && n.depth > body.depth && (nodes.find((x) => x.name === 'Body' && x.at > body.at) === undefined || true));
const firstRect = inBody.find((n) => n.name === 'Rectangle');
console.log('container:', firstRect?.attr);
const kids = inBody.filter((n) => items.has(n.name) && n.parent && n.parent.name === 'ReportItems' && n.parent.parent === firstRect);
console.log(`items directly in the container (${kids.length}), in file order:`);
for (const k of kids) {
  const topNode = nodes.find((n) => n.name === 'Top' && n.parent === k);
  const top = topNode ? /<Top>([^<]*)<\/Top>/.exec(xml.slice(topNode.at))?.[1] : '-';
  const own = (tag) => { const t = nodes.find((n) => n.name === tag && n.parent === k); return t ? (new RegExp(`<${tag}>([^<]*)</${tag}>`).exec(xml.slice(t.at))?.[1] ?? 'set') : ''; };
  const z = own('ZIndex');
  const keep = own('KeepTogether');
  const hid = nodes.some((n) => n.name === 'Visibility' && n.parent === k) ? ' hidden-rule' : '';
  console.log(`  ${k.attr} top=${top}${z ? ' z=' + z : ''}${keep ? ' keep=' + keep : ''}${hid}`);
}
const stem = (firstRect?.attr ?? '').replace(/_[A-Za-z]+_?\d+$/, '');
console.log('stem:', JSON.stringify(stem));
const pattern = new RegExp(`^${stem.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}_([A-Za-z]+_?)(\\d+)$`);
const named = nodes.filter((n) => n.attr);
const matching = named.filter((n) => pattern.test(n.attr));
const count = {};
for (const n of matching) { const k = `${n.name}:${pattern.exec(n.attr)[1]}`; count[k] = (count[k] ?? 0) + 1; }
console.log('names of that form, by kind:', JSON.stringify(count));
const seen = {};
for (const n of named) (seen[n.attr] ??= []).push(n.name);
const twice = Object.entries(seen).filter(([, v]) => v.length > 1 && v.some((t) => items.has(t)));
console.log('names used more than once:', twice.length, JSON.stringify(twice.slice(0, 8)));
const others = matching.filter((n) => !items.has(n.name) || n.at < body.at);
console.log('non-item or out-of-body elements with item-like names:', others.length, JSON.stringify(others.slice(0, 8).map((n) => `${n.name}:${n.attr}`)));
console.log('Body count:', nodes.filter((n) => n.name === 'Body').length, 'ReportSection count:', nodes.filter((n) => n.name === 'ReportSection').length);
