 return false;

if (hc.length !== n || !empty(hc[0]!) || empty(hc[1]!)) {
    if (hc.length !== n) review.push({ item: tablix.attributes.Name ?? 'Table', message: `its heading row has ${hc.length} cells for ${n} columns, so its first column was not folded` });
